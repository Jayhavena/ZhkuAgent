param([switch]$CheckOnly)

function Credential([string]$name) {
 foreach($scope in @('Process','User','Machine')) {$value=[Environment]::GetEnvironmentVariable($name,$scope);if(![string]::IsNullOrWhiteSpace($value)){return $value}}
 throw ('Credential unavailable: '+$name)
}
function Child([string]$exe,[string[]]$childArgs,[int]$seconds,[hashtable]$childEnv=@{}) {
 $info=[Diagnostics.ProcessStartInfo]::new();$info.FileName=$exe;$info.UseShellExecute=$false;$info.CreateNoWindow=$true
 $info.RedirectStandardOutput=$true;$info.RedirectStandardError=$true;$info.StandardOutputEncoding=[Text.UTF8Encoding]::new($false);$info.StandardErrorEncoding=$info.StandardOutputEncoding
 foreach($arg in $childArgs){$info.ArgumentList.Add($arg)}
 foreach($name in $childEnv.Keys){$info.Environment[$name]=$childEnv[$name]}
 foreach($name in @('ZHKU_RULECHECK_TASK_BATCH_DIRECTORY','ZHKU_RULECHECK_TASK_PACKAGE_SHA256','ZHKU_RULECHECK_TASK_SUMMARY_SHA256')){$info.Environment.Remove($name)|Out-Null}
 $process=[Diagnostics.Process]::new();$process.StartInfo=$info
 try {
  $process.Start()|Out-Null;$stdout=$process.StandardOutput.ReadToEndAsync();$stderr=$process.StandardError.ReadToEndAsync()
  $timedOut=!$process.WaitForExit($seconds*1000)
  if($timedOut){$process.Kill($true);$process.WaitForExit(5000)|Out-Null}
  $closed=[Threading.Tasks.Task]::WaitAll([Threading.Tasks.Task[]]@($stdout,$stderr),2000)
  if(!$closed){throw 'Owned child output did not close'}
  $result=[pscustomobject]@{exitCode=$process.ExitCode;timedOut=$timedOut;resourcesClosed=$process.HasExited;output=$stdout.Result;errorOutput=$stderr.Result}
  foreach($name in @('ZHKU_DB_PASSWORD','DASHSCOPE_API_KEY','POSTGRES_PASSWORD')){if($childEnv.ContainsKey($name)){$secret=$childEnv[$name];if($secret){$result.output=$result.output.Replace($secret,'[REDACTED]');$result.errorOutput=$result.errorOutput.Replace($secret,'[REDACTED]')}}}
  return $result
 } finally {if($process.Id-and !$process.HasExited){$process.Kill($true)};$process.Dispose()}
}
function RequireSuccess($result){if($result.exitCode-ne 0-or $result.timedOut-or !$result.resourcesClosed){throw ($result.output+$result.errorOutput)}}
if((Get-FileHash -LiteralPath (Join-Path $root $package)).Hash.ToLowerInvariant()-ne $sha){throw 'Package SHA mismatch'}
if(Test-Path -LiteralPath (Join-Path $base 'execution-reserved.json')){throw 'Batch already reserved'}
$java=(Get-Command java.exe).Source;$docker=(Get-Command docker.exe).Source

$checked=Child $java ($javaPrefix+@('check',$root,$package,$sha)) 20;RequireSuccess $checked

& (Join-Path $root 'knowledge/scripts/ensure-zhku-postgres.cmd') -DockerOnly
if($LASTEXITCODE-ne 0){throw 'Docker readiness failed'}

$out=$relative+'/live-'+[guid]::NewGuid().ToString('N')
try {

 $created=Child $docker @('run','--rm','-d','--pull','never','--name',$container,'--publish','127.0.0.1:55519:5432','--env','POSTGRES_PASSWORD','--env','POSTGRES_USER=zhkuagent','--env','POSTGRES_DB=zhkuagent_rulecheck_e2e','pgvector/pgvector:0.8.6-pg16-bookworm') 15 @{POSTGRES_PASSWORD=$password};RequireSuccess $created
 $ready=$false
 for($i=0;$i-lt 20;$i++) {$null=Credential 'ZHKU_DB_PASSWORD';$probe=Child $docker @('exec',$container,'pg_isready','-U','zhkuagent','-d','zhkuagent_rulecheck_e2e') 10;if($probe.exitCode-eq 0-and !$probe.timedOut){$ready=$true;break};Start-Sleep -Milliseconds 250}
 if(!$ready){throw 'Dedicated temporary database not ready'}
 $environment=@{ZHKU_DB_PASSWORD=(Credential 'ZHKU_DB_PASSWORD');DASHSCOPE_API_KEY=(Credential 'DASHSCOPE_API_KEY');ZHKU_DB_URL='jdbc:postgresql://127.0.0.1:55519/zhkuagent_rulecheck_e2e';ZHKU_RULE_AGENT_MODE='ONLINE';ZHKU_RULE_AGENT_AUTHORIZATION_ID=$sha}
 $result=Child $java ($javaPrefix+@('run',$root,$package,$sha,$out)) 1350 $environment
 $directory=Join-Path $root $out;[IO.Directory]::CreateDirectory($directory)|Out-Null

} finally {
 $cleaned=Child $docker @('rm','-f',$container) 15

 RequireSuccess $cleaned
}
